# Python dependencies of the register generator (`regtool.py`, shipped in the
# `register_interface` Bender dependency and derived from OpenTitan's `reggen`).
#
# They are only needed by `make regen_regs`: building and simulating the IP
# does not use Python at all.
#
# Every package (including the transitive MarkupSafe) is pinned to an exact
# version so that the environment is identical on every machine. The pins are
# installable on Python 3.8 up to 3.12 (checked on 3.8, 3.11 and 3.12).

# Reads the register description (`src/registers/axi_tmu_<variant>_regs.hjson`)
hjson==3.1.0

# Template engine that renders the generated RTL (`*_reg_pkg.sv`, `*_reg_top.sv`)
Mako==1.3.12
MarkupSafe==2.1.5

# Imported at module level by reggen (generators for DV / FPV collateral)
PyYAML==6.0.3

# Imported at module level by reggen (markdown documentation helpers)
tabulate==0.9.0

# reggen imports `pkg_resources` to locate its templates. Python >= 3.12 venvs
# do not ship setuptools, and recent setuptools releases (84.0.0 was tried) no
# longer contain `pkg_resources`, so it has to be installed explicitly and
# kept on an old release. 75.3.4 is the last release that supports Python 3.8.
setuptools==75.3.4
